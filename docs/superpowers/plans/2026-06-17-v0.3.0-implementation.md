```ts
function matchesMimeType(ruleType: string, fileType: string): boolean {
  const normalRule = ruleType.toLowerCase()
  const normalFile = fileType.toLowerCase()
  if (normalRule.endsWith('/*')) {
    return normalFile.startsWith(normalRule.slice(0, -1))
  }
  return normalRule === normalFile
}
```

```ts
function matchesWildcardPattern(pattern: string, path: string): string[] | null {
  const patternParts = pattern.split('.')
  const pathParts = path.split('.')
  if (patternParts.length !== pathParts.length) return null
  const indices: string[] = []
  for (let i = 0; i < patternParts.length; i++) {
    if (patternParts[i] === '*') {
      if (!/^\d+$/.test(pathParts[i])) return null
      indices.push(pathParts[i])
    } else if (patternParts[i] !== pathParts[i]) {
      return null
    }
  }
  return indices
}

function resolveWildcardDependents(dependentPatterns: string[], indices: string[]): string[] {
  return dependentPatterns.map(dep => {
    let i = 0
    return dep.replace(/\*/g, () => indices[i++] ?? '*')
  })
}
```

```ts
export interface WildcardDependency {
  pattern: string
  dependents: string[]
}

export function compileDependencyScopes(
  dependencies: Record<string, string[] | undefined>,
  initialValues: any
): { preComputedScopes: Record<string, string[]>; wildcardDependencies: WildcardDependency[] } {
  const preComputedScopes: Record<string, string[]> = {}
  const wildcardDependencies: WildcardDependency[] = []
  const allFieldPaths = extractAllPaths(initialValues)

  const isWildcardEntry = (key: string) => key.includes('*')

  // Separate wildcard from static
  const staticDependencies: Record<string, string[] | undefined> = {}
  Object.entries(dependencies).forEach(([key, val]) => {
    if (isWildcardEntry(key) || (val ?? []).some(v => v.includes('*'))) {
      wildcardDependencies.push({ pattern: key, dependents: val ?? [] })
    } else {
      staticDependencies[key] = val
    }
  })

  const resolveTransitiveClosure = (currentPath: string, visited: Set<string>) => {
    if (visited.has(currentPath)) return
    visited.add(currentPath)
    const directDependents = staticDependencies[currentPath]
    if (directDependents) {
      for (const dep of directDependents) resolveTransitiveClosure(dep, visited)
    }
  }

  allFieldPaths.forEach(path => {
    const visited = new Set<string>()
    resolveTransitiveClosure(path, visited)
    preComputedScopes[path] = Array.from(visited)
  })

  Object.keys(staticDependencies).forEach(path => {
    if (!preComputedScopes[path]) {
      const visited = new Set<string>()
      resolveTransitiveClosure(path, visited)
      preComputedScopes[path] = Array.from(visited)
    }
  })

  return { preComputedScopes, wildcardDependencies }
}
```

```ts
  paths: Path<T> | string | Array<Path<T> | string>,
  callback: (values: Record<string, unknown>) => void
): (() => void) => {
  const pathArray = (Array.isArray(paths) ? paths : [paths]) as string[]
  const uniquePaths = [...new Set(pathArray)]

  if (uniquePaths.length === 0) return () => {}

  let pending = false

  const fire = () => {
    if (pending) return
    pending = true
    // Use microtask to batch multiple-path changes in same tick
    queueMicrotask(() => {
      pending = false
      const snapshot: Record<string, unknown> = {}
      uniquePaths.forEach(p => { snapshot[p] = getNestedValue(values, p) })
      try { callback(snapshot) } catch (err) {
        console.error('[NeutroForm] watch callback threw:', err)
      }
    })
  }

  const teardowns: Array<() => void> = []
  let tornDown = false

  uniquePaths.forEach(p => {
    const unsub = subscribeToPath(p as Path<T>, fire as any)
    teardowns.push(unsub)
  })

  return () => {
    if (tornDown) return
    tornDown = true
    teardowns.forEach(u => u())
  }
}
```

```ts
  form: FormInstance<T>,
  paths: Path<T> | string | Array<Path<T> | string>
): Record<string, unknown> {
  const pathArray = Array.isArray(paths) ? paths : [paths]
  const [watched, setWatched] = React.useState<Record<string, unknown>>(() => {
    const snap: Record<string, unknown> = {}
    pathArray.forEach(p => { snap[p as string] = (form.get as any)(p) })

  })

    return stop
  }, [form, JSON.stringify(pathArray)])

  return watched
}
```

```ts
import { ref, readonly, onUnmounted, watchEffect, type MaybeRef, isRef, unref } from 'vue'

export function useVueWatch<T extends object>(
  form: FormInstance<T>,
  paths: MaybeRef<Array<Path<T> | string>> | Path<T> | string
): { [key: string]: ReturnType<typeof readonly<ReturnType<typeof ref<unknown>>>> } {
  const pathArray = computed(() => {
    const raw = isRef(paths) ? unref(paths) : (Array.isArray(paths) ? paths : [paths])
    return raw as string[]
  })

  const watched = ref<Record<string, unknown>>({})

  let stop: (() => void) | null = null

  const resubscribe = () => {
    if (stop) stop()
    const current = pathArray.value
    const snap: Record<string, unknown> = {}
    current.forEach(p => { snap[p] = (form.get as any)(p) })
    watched.value = snap
    stop = form.watch(current, (vals) => { watched.value = { ...vals } })
  }

  watchEffect(resubscribe)
  onUnmounted(() => { if (stop) stop() })

  return readonly(watched) as any
}
```

```ts
  onSubmitCallback: (payload: Partial<T>) => void | Promise<void>
```

```ts
const focusFirstError = (): boolean => {
  const errorPaths = Object.keys(errors)
  if (errorPaths.length === 0) return false

  const connected = errorPaths
    .map(p => {
      const ref = connectionRegistry.get(p)
      if (!ref) return null
      const el = ref.deref()
      if (!el || !el.isConnected) return null

  try {
    connected[0].el.focus()
  } catch (err) {

    return false
  }
  return true
}
```

```ts
/**
 * Path validation trie for dev-only runtime unknown-path detection.
 * Released in 0.4.0 if benchmarks pass; prototype only in 0.3.0.
 */

type TrieNode = {
  isLeaf: boolean
  isArray: boolean
  children: Map<string, TrieNode>
}

const ARRAY_WILDCARD = '[*]'

function newNode(): TrieNode {
  return { isLeaf: false, isArray: false, children: new Map() }
}

export function buildPathTrie(obj: unknown, node: TrieNode = newNode()): TrieNode {
  if (obj === null || typeof obj !== 'object') {
    node.isLeaf = true
    return node
  }
  if (Array.isArray(obj)) {
    node.isArray = true
    const itemTemplate = obj[0] ?? {}
    let child = node.children.get(ARRAY_WILDCARD)
    if (!child) { child = newNode(); node.children.set(ARRAY_WILDCARD, child) }
    buildPathTrie(itemTemplate, child)
    return node
  }
  for (const [key, val] of Object.entries(obj)) {
    let child = node.children.get(key)
    if (!child) { child = newNode(); node.children.set(key, child) }
    buildPathTrie(val, child)
  }
  return node
}

export function isKnownPath(trie: TrieNode, path: string): boolean {
  const parts = path.split('.')
  let node: TrieNode | undefined = trie
  for (const part of parts) {
    if (!node) return false
    if (node.isArray) {
      const wildcard = node.children.get(ARRAY_WILDCARD)
      if (!wildcard) return false
      if (/^\d+$/.test(part)) {
        node = wildcard
        continue
      }
    }
    node = node.children.get(part)
  }
  return node !== undefined
}
```

```ts
// Inside createForm, after state variables:
const computedFields = config.computed ?? {}
const computedKeys = Object.keys(computedFields) as Array<keyof T>

const runComputedPass = () => {
  if (computedKeys.length === 0) return
  let changed = false
  batch(() => {
    for (const key of computedKeys) {
      const fn = computedFields[key]!
      const newVal = fn(values)
      const currentVal = (values as any)[key]
      if (newVal !== currentVal) {

      }

  })

}
```
